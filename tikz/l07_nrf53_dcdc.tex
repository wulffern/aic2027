% The DC/DC pins of the nRF5340, simplified from the reference schematic
% in the product specification. Each of the three inductors gives away an
% internal DC/DC converter. (1) DCC switches through L2 onto DECR and
% DECRF, the radio supply, decoupled by C3. (2) DCCD switches through L3
% onto DECD, the digital supply, decoupled by C5. (3) DCCH switches
% through L4 back onto VDD: the high voltage converter takes VDDH (up to
% 5.5 V) down to VDD for the rest of the chip. The VDD decoupling
% capacitors, crystal, antenna and debug pins of the original are left
% out because they say nothing about the regulators. Redrawn so the book
% does not carry the datasheet figure.

\begin{circuitikz}[american, thick, transform shape, circuit ee IEC]

  \input{tikz/ckt_lib.tex}

  \tikzset{pin/.style={draw, fill=white, minimum size=0.18cm, inner sep=0pt}}

  % chip
  \draw (5.0,0.6) rectangle (7.6,13.0);
  \node[rotate=90] at (7.0,6.8) {nRF5340};

  \foreach \y/\n in {12.2/VDD, 11.6/DCC, 10.2/DECR, 9.6/DECRF,
                     7.8/VDD, 7.2/DCCD, 5.8/DECD,
                     3.8/VDD, 2.4/DCCH, 1.4/VDDH} {
    \node[pin] at (5.0,\y) {};
    \node[anchor=west] at (5.15,\y) {\small \n};
  }

  % group 1: DCC through L2 onto DECR and DECRF
  \draw (1.2,12.2) -- (4.91,12.2);
  \draw (1.2,12.2) \vsupply;
  \node[anchor=south] at (1.2,12.6) {\small VDD\_nRF};
  \draw (4.91,11.6) -- (3.0,11.6) to [L, l=L2] (3.0,10.2);
  \draw (3.0,10.2) -- (4.91,10.2);
  \fill (3.0,10.2) circle (0.075);
  \draw (4.91,9.6) -- (4.2,9.6) -- (4.2,10.2);
  \fill (4.2,10.2) circle (0.075);
  \draw (3.0,8.7) \vcapacitor{C3};
  \draw (3.0,8.7) \vground;

  % group 2: DCCD through L3 onto DECD
  \draw (1.2,7.8) -- (4.91,7.8);
  \draw (1.2,7.8) \vsupply;
  \node[anchor=south] at (1.2,8.2) {\small VDD\_nRF};
  \draw (4.91,7.2) -- (3.0,7.2) to [L, l=L3] (3.0,5.8);
  \draw (3.0,5.8) -- (4.91,5.8);
  \fill (3.0,5.8) circle (0.075);
  \draw (3.0,4.3) \vcapacitor{C5};
  \draw (3.0,4.3) \vground;

  % group 3: DCCH through L4 back onto VDD, VDDH from the high supply
  \draw (1.2,3.8) -- (4.91,3.8);
  \draw (1.2,3.8) \vsupply;
  \node[anchor=south] at (1.2,4.2) {\small VDD\_nRF};
  \draw (4.91,2.4) -- (3.0,2.4) to [L, l_=L4] (3.0,3.8);
  \fill (3.0,3.8) circle (0.075);
  \draw (1.2,1.4) -- (4.91,1.4);
  \draw (1.2,1.4) \vsupply;
  \node[anchor=south] at (1.2,1.8) {\small VDD\_HV};

  % the three DC/DC inductors
  \foreach \y/\n in {10.9/1, 6.5/2, 3.1/3} {
    \draw[red, very thick, ->] (1.0,\y-0.3) to [bend left=20] (2.55,\y);
    \node[red, anchor=east] at (1.0,\y-0.3) {\large\bfseries \n};
  }

\end{circuitikz}
