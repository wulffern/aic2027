% The regulators inside the nRF5340, after the block diagram in the
% product specification. Three supply pins come in on the left: VDD,
% the high voltage VDDH and the USB VBUS. VDDH goes through the high
% voltage regulator VREGH, whose output joins VDD on one internal rail.
% That rail feeds the GPIOs directly, the radio regulator VREGRADIO that
% supplies the network core, and the main regulator VREGMAIN that
% supplies the application core. VBUS has its own USB regulator USBREG,
% which supplies the application core's USB block. Redrawn so the book
% does not carry the datasheet figure.

\begin{tikzpicture}[thick]

  \tikzset{blk/.style={draw, align=center, minimum height=0.9cm}}

  % chip outline
  \draw (0,0) rectangle (9.6,5.6);

  % pins
  \foreach \y/\n in {4.2/VDD, 2.7/VDDH, 1.0/VBUS} {
    \draw[fill=white] (-0.1,\y-0.1) rectangle (0.1,\y+0.1);
    \node[anchor=east] at (-0.15,\y) {\n};
  }

  % regulators on the pins
  \node[blk, minimum width=2.2cm] (vregh) at (2.0,2.7) {VREGH};
  \node[blk, minimum width=2.2cm] (usb)   at (2.0,1.0) {USBREG};
  \draw[->] (0.1,2.7) -- (vregh.west);
  \draw[->] (0.1,1.0) -- (usb.west);

  % internal VDD rail
  \draw (0.1,4.2) -- (3.6,4.2);
  \draw (vregh.east) -- (3.6,2.7);
  \draw (3.6,2.0) -- (3.6,4.9) -- (7.0,4.9) coordinate (gpioin);
  \fill (3.6,4.2) circle (0.075);
  \fill (3.6,2.7) circle (0.075);

  % GPIOs sit on the chip edge
  \draw[dashed, fill=white] (6.2,5.35) rectangle (7.8,5.85);
  \node at (7.0,5.6) {\small GPIOs};
  \draw[->] (gpioin) -- (7.0,5.35);

  % radio and main regulators
  \node[blk, minimum width=2.4cm] (vrad)  at (5.2,3.6) {VREGRADIO};
  \node[blk, minimum width=2.4cm] (vmain) at (5.2,2.0) {VREGMAIN};
  \draw[->] (3.6,3.6) -- (vrad.west);
  \fill (3.6,3.6) circle (0.075);
  \draw[->] (3.6,2.0) -- (vmain.west);

  % cores
  \node[draw, minimum width=2.4cm, minimum height=1.1cm] (net) at (8.1,3.6) {Network core};
  \draw (6.9,0.4) rectangle (9.3,2.6);
  \node at (8.1,1.5) {Application core};
  \draw[->] (vrad.east) -- (net.west);
  \draw[->] (vmain.east) -- (6.9,2.0);
  \draw[->] (usb.east) -- (6.9,1.0);

\end{tikzpicture}
